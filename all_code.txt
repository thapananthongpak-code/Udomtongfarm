import { useMemo, useState } from "react";
import { Link, useLocation, useNavigate } from "react-router-dom";
import {
  sendEmailVerification,
  signInWithEmailAndPassword,
  signOut,
} from "firebase/auth";
import { auth } from "../../services/firebase";
import { getRoleByEmail } from "../../utils/roles";
function useQuery() {
  const { search } = useLocation();
  return useMemo(() => new URLSearchParams(search), [search]);
}

export default function VerifyEmail() {
  const navigate = useNavigate();
  const query = useQuery();

  const emailFromQuery = query.get("email") ?? "";
  const [email, setEmail] = useState(emailFromQuery);
  const [password, setPassword] = useState("");

  const [loading, setLoading] = useState(false);
  const [msg, setMsg] = useState<string>("");
  const [err, setErr] = useState<string>("");

  async function resendToLoggedInUser() {
    setErr("");
    setMsg("");
    setLoading(true);

    try {
      const user = auth.currentUser;
      if (!user) {
        setErr("ยังไม่ได้ login — ถ้าต้องการ resend ให้กรอก Email+Password ด้านล่าง");
        return;
      }

      await sendEmailVerification(user);
      setMsg("ส่งอีเมลยืนยันใหม่แล้ว ✅ เช็ค inbox / spam");
    } catch (e: any) {
      setErr(e?.message ?? "Resend failed");
    } finally {
      setLoading(false);
    }
  }

  async function loginToResend(e: React.FormEvent) {
    e.preventDefault();
    setErr("");
    setMsg("");
    setLoading(true);

    try {
      const cred = await signInWithEmailAndPassword(auth, email, password);

      if (cred.user.emailVerified) {
        // verified แล้ว → เข้าได้เลย
        const role = getRoleByEmail(cred.user.email);
        navigate(role === "admin" ? "/admin" : "/encyclopedia", { replace: true });
        return;
      }

      await sendEmailVerification(cred.user);
      setMsg("ส่งอีเมลยืนยันใหม่แล้ว ✅ เช็ค inbox / spam");

      // กันหลุดไปหน้าอื่นทั้งที่ยังไม่ verify
      await signOut(auth);
    } catch (e: any) {
      setErr(e?.message ?? "Login/Resend failed");
    } finally {
      setLoading(false);
    }
  }

  async function checkVerified() {
    setErr("");
    setMsg("");
    setLoading(true);

    try {
      const user = auth.currentUser;
      if (!user) {
        setErr("ยังไม่ได้ login — กรุณา login ก่อนเพื่อเช็คสถานะ");
        return;
      }

      await user.reload();

      if (!user.emailVerified) {
        setErr("อีเมลยังไม่ยืนยัน ❌ ลองเช็คเมลแล้วกดลิงก์ก่อน แล้วกลับมากดปุ่มนี้อีกที");
        return;
      }

      const role = getRoleByEmail(user.email);
      navigate(role === "admin" ? "/admin" : "/encyclopedia", { replace: true });
    } catch (e: any) {
      setErr(e?.message ?? "Check failed");
    } finally {
      setLoading(false);
    }
  }

  return (
    <div style={{ maxWidth: 760, margin: "0 auto", padding: "24px 16px" }}>
      <h1 style={{ fontSize: 50, margin: "18px 0" }}>Verify your email</h1>

      <p style={{ fontSize: 18, color: "#334155", lineHeight: 1.6 }}>
        เราส่งลิงก์ยืนยันไปที่อีเมลของคุณแล้ว  
        กรุณาเปิดอีเมล แล้วกดลิงก์เพื่อยืนยันก่อนเข้าใช้งาน
      </p>

      <div style={{ marginTop: 10, fontSize: 16, color: "#64748b" }}>
        Tips: เช็ค <b>Spam / Junk</b> ด้วยนะ
      </div>

      {(msg || err) && (
        <div
          style={{
            marginTop: 14,
            padding: "12px 14px",
            borderRadius: 12,
            border: "1px solid #e2e8f0",
            background: msg ? "rgba(34,197,94,0.08)" : "rgba(239,68,68,0.06)",
            color: msg ? "#166534" : "#991b1b",
            fontWeight: 700,
          }}
        >
          {msg || err}
        </div>
      )}

      <div style={{ display: "flex", gap: 10, flexWrap: "wrap", marginTop: 16 }}>
        <button
          type="button"
          onClick={resendToLoggedInUser}
          disabled={loading}
          style={{
            padding: "12px 14px",
            borderRadius: 12,
            border: "1px solid #ddd",
            background: "#fff",
            fontSize: 16,
            cursor: loading ? "not-allowed" : "pointer",
            fontWeight: 800,
          }}
        >
          {loading ? "Working..." : "Resend (if you are logged in)"}
        </button>

        <button
          type="button"
          onClick={checkVerified}
          disabled={loading}
          style={{
            padding: "12px 14px",
            borderRadius: 12,
            border: "1px solid #ddd",
            background: "#0f172a",
            color: "#fff",
            fontSize: 16,
            cursor: loading ? "not-allowed" : "pointer",
            fontWeight: 900,
          }}
        >
          {loading ? "Checking..." : "I verified → Continue"}
        </button>
      </div>

      <div
        style={{
          marginTop: 18,
          padding: 14,
          borderRadius: 14,
          border: "1px solid #e2e8f0",
          background: "#fff",
        }}
      >
        <div style={{ fontWeight: 900, marginBottom: 10 }}>
          Resend by logging in (Email + Password)
        </div>

        <form onSubmit={loginToResend} style={{ display: "grid", gap: 12 }}>
          <label>
            <div style={{ marginBottom: 6 }}>Email</div>
            <input
              value={email}
              onChange={(e) => setEmail(e.target.value)}
              type="email"
              required
              placeholder="name@email.com"
              style={{
                width: "100%",
                padding: "12px 14px",
                borderRadius: 10,
                border: "1px solid #ccc",
                fontSize: 16,
              }}
            />
          </label>

          <label>
            <div style={{ marginBottom: 6 }}>Password</div>
            <input
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              type="password"
              required
              placeholder="Your password"
              style={{
                width: "100%",
                padding: "12px 14px",
                borderRadius: 10,
                border: "1px solid #ccc",
                fontSize: 16,
              }}
            />
          </label>

          <button
            disabled={loading}
            style={{
              marginTop: 4,
              padding: "12px 14px",
              borderRadius: 12,
              border: "1px solid #ddd",
              background: loading ? "#f3f4f6" : "#fff",
              fontSize: 18,
              cursor: loading ? "not-allowed" : "pointer",
              fontWeight: 800,
            }}
          >
            {loading ? "Working..." : "Login & Resend verification"}
          </button>
        </form>

        <div style={{ marginTop: 12, color: "#2563eb", fontSize: 16 }}>
          <Link to="/login">Back to Login</Link>
        </div>
      </div>
    </div>
  );
}

import { useMemo, useRef, useState } from "react";
import { useAuth } from "../store/AuthContext";
import { useSpeciesStore } from "../store/speciesStore";

const STORAGE_KEY = "udomtong_admin_whitelist_v1";

function loadExtraAdmins(): string[] {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    const arr = raw ? (JSON.parse(raw) as string[]) : [];
    return Array.isArray(arr) ? arr : [];
  } catch {
    return [];
  }
}

function saveExtraAdmins(emails: string[]) {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(emails));
}

export default function AdminSettings() {
  const { user } = useAuth();
  const [input, setInput] = useState("");

  const [extras, setExtras] = useState<string[]>(() => loadExtraAdmins());

  const normalizedExtras = useMemo(
    () =>
      Array.from(new Set(extras.map((e) => e.toLowerCase().trim()))).filter(
        Boolean
      ),
    [extras]
  );

  function addEmail() {
    const email = input.toLowerCase().trim();
    if (!email) return;
    setExtras((prev) => {
      const next = Array.from(new Set([...prev, email]));
      saveExtraAdmins(next);
      return next;
    });
    setInput("");
  }

  function removeEmail(email: string) {
    setExtras((prev) => {
      const next = prev.filter((e) => e.toLowerCase().trim() !== email);
      saveExtraAdmins(next);
      return next;
    });
  }

  // ✅ Backup/Restore (Species)
  const exportJSON = useSpeciesStore((s) => s.exportJSON);
  const importJSON = useSpeciesStore((s) => s.importJSON);
  const mergeImportJSON = useSpeciesStore((s: any) => s.mergeImportJSON); // ✅ merge (ถ้า types ยังไม่อัปเดต)

  const fileRestoreRef = useRef<HTMLInputElement | null>(null);
  const fileMergeRef = useRef<HTMLInputElement | null>(null);

  function downloadBackup() {
    const text = exportJSON();
    const blob = new Blob([text], { type: "application/json" });
    const url = URL.createObjectURL(blob);

    const a = document.createElement("a");
    a.href = url;
    a.download = `udomtong-species-backup-${new Date()
      .toISOString()
      .slice(0, 10)}.json`;
    a.click();

    URL.revokeObjectURL(url);
  }

  async function handleImportFile(file: File) {
    const ok = confirm("นำเข้าไฟล์จะ “ทับข้อมูล species เดิมทั้งหมด” แน่ใจนะ?");
    if (!ok) return;

    const text = await file.text();
    const res = importJSON(text);

    if (!res.ok) {
      alert("Import ไม่สำเร็จ: " + (res.error || ""));
      return;
    }

    alert(`Import สำเร็จ: ${res.count ?? 0} รายการ`);
  }

  async function handleMergeImportFile(file: File) {
    const ok = confirm(
      "นำเข้าแบบ MERGE จะ “รวมเข้ากับของเดิม” (ไม่ทับ) — ถ้า id ซ้ำใน type เดียวกันจะข้ามอัตโนมัติ แน่ใจนะ?"
    );
    if (!ok) return;

    if (typeof mergeImportJSON !== "function") {
      alert("ยังไม่มี mergeImportJSON ใน speciesStore.ts");
      return;
    }

    const text = await file.text();
    const res = mergeImportJSON(text);

    if (!res?.ok) {
      alert("Merge import ไม่สำเร็จ: " + (res?.error || ""));
      return;
    }

    const added = res.added ?? 0;
    const skipped = res.skipped ?? 0;
    const invalid = res.invalid ?? 0;

    alert(
      `Merge import สำเร็จ ✅\n- เพิ่มใหม่: ${added}\n- ข้าม (ซ้ำ): ${skipped}\n- ไม่ผ่าน validate: ${invalid}`
    );
  }

  return (
    <div style={{ maxWidth: 900 }}>
      <h1 style={{ fontSize: 34, margin: 0, fontWeight: 900 }}>
        Admin settings
      </h1>
      <p style={{ color: "#475569", marginTop: 6 }}>
        โหมดฟรี: เก็บ “admin เพิ่มเติม” ไว้ใน localStorage (ชั่วคราว) — admin หลักยังเป็น
        <b> thapananthongpak@gmail.com</b>
      </p>

      <div
        style={{
          marginTop: 14,
          background: "#fff",
          border: "1px solid #e2e8f0",
          borderRadius: 16,
          padding: 16,
        }}
      >
        <div style={{ fontWeight: 900, marginBottom: 10 }}>Add admin email</div>

        <div style={{ display: "flex", gap: 10, flexWrap: "wrap" }}>
          <input
            value={input}
            onChange={(e) => setInput(e.target.value)}
            placeholder="admin2@email.com"
            style={{
              width: 360,
              maxWidth: "100%",
              padding: "10px 12px",
              borderRadius: 12,
              border: "1px solid #e2e8f0",
            }}
          />
          <button
            onClick={addEmail}
            style={{
              padding: "10px 14px",
              borderRadius: 12,
              border: "1px solid #e2e8f0",
              background: "#0f172a",
              color: "#fff",
              fontWeight: 900,
              cursor: "pointer",
            }}
          >
            Add
          </button>
        </div>

        <div style={{ marginTop: 14, color: "#64748b", fontSize: 12 }}>
          Logged in as: <b>{user?.email ?? "-"}</b>
        </div>
      </div>

      <div
        style={{
          marginTop: 14,
          background: "#fff",
          border: "1px solid #e2e8f0",
          borderRadius: 16,
          overflow: "hidden",
        }}
      >
        <div
          style={{
            padding: "12px 14px",
            background: "#f8fafc",
            fontWeight: 900,
            borderBottom: "1px solid #e2e8f0",
          }}
        >
          Extra admins (localStorage)
        </div>

        {normalizedExtras.length === 0 ? (
          <div style={{ padding: 14, color: "#64748b" }}>No extra admins yet.</div>
        ) : (
          normalizedExtras.map((email) => (
            <div
              key={email}
              style={{
                display: "flex",
                justifyContent: "space-between",
                padding: "12px 14px",
                borderBottom: "1px solid #f1f5f9",
              }}
            >
              <div style={{ fontWeight: 800 }}>{email}</div>
              <button
                onClick={() => removeEmail(email)}
                style={{
                  padding: "6px 10px",
                  borderRadius: 10,
                  border: "1px solid #e2e8f0",
                  background: "#fff",
                  cursor: "pointer",
                  fontWeight: 900,
                }}
              >
                Remove
              </button>
            </div>
          ))
        )}
      </div>

      {/* ✅ Species Backup/Restore */}
      <div
        style={{
          marginTop: 14,
          background: "#fff",
          border: "1px solid #e2e8f0",
          borderRadius: 16,
          padding: 16,
        }}
      >
        <div style={{ fontWeight: 900, marginBottom: 6 }}>Species backup (JSON)</div>
        <div style={{ color: "#64748b", fontSize: 12 }}>
          กันข้อมูลหาย: export เก็บไว้ / import เพื่อ restore (โหมดฟรี ไม่ใช้ Firestore)
        </div>

        <div style={{ display: "flex", gap: 10, flexWrap: "wrap", marginTop: 12 }}>
          <button
            onClick={downloadBackup}
            style={{
              padding: "10px 14px",
              borderRadius: 12,
              border: "1px solid #e2e8f0",
              background: "#0f172a",
              color: "#fff",
              fontWeight: 900,
              cursor: "pointer",
            }}
          >
            Export JSON
          </button>

          <button
            onClick={() => fileRestoreRef.current?.click()}
            style={{
              padding: "10px 14px",
              borderRadius: 12,
              border: "1px solid #e2e8f0",
              background: "#fff",
              color: "#0f172a",
              fontWeight: 900,
              cursor: "pointer",
            }}
          >
            Import JSON (restore)
          </button>

          <button
            onClick={() => fileMergeRef.current?.click()}
            style={{
              padding: "10px 14px",
              borderRadius: 12,
              border: "1px solid #e2e8f0",
              background: "#fff",
              color: "#0f172a",
              fontWeight: 900,
              cursor: "pointer",
            }}
          >
            Import JSON (merge)
          </button>

          {/* Restore input */}
          <input
            ref={fileRestoreRef}
            type="file"
            accept="application/json"
            style={{ display: "none" }}
            onChange={(e) => {
              const file = e.target.files?.[0];
              if (!file) return;
              handleImportFile(file);
              e.currentTarget.value = "";
            }}
          />

          {/* Merge input */}
          <input
            ref={fileMergeRef}
            type="file"
            accept="application/json"
            style={{ display: "none" }}
            onChange={(e) => {
              const file = e.target.files?.[0];
              if (!file) return;
              handleMergeImportFile(file);
              e.currentTarget.value = "";
            }}
          />
        </div>

        <div style={{ marginTop: 10, color: "#64748b", fontSize: 12 }}>
          *Restore จะ “ทับข้อมูล species เดิมทั้งหมด” (มี confirm ก่อน) <br />
          *Merge จะ “รวมเข้าไป” โดยไม่ทับ: ถ้า id ซ้ำใน type เดียวกันจะข้าม
        </div>
      </div>

      <div style={{ marginTop: 14, color: "#64748b", fontSize: 12 }}>
        *หมายเหตุ: แบบโปรดักชันจริง จะย้าย whitelist ไป Firestore/Custom Claims ทีหลัง
      </div>
    </div>
  );
}

type RefRow = { title: string; url: string };

function emptySpecies(type: SpeciesType): Species {
  return {
    id: "",
    type,
    name_th: "",
    name_en: "",
    scientific_name: "",
    short_description: "",
    description: "",
    image: type === "animal" ? "/images/animals/" : "/images/plants/",
    tags: [],
    references: [{ title: "Wikipedia", url: "" }],
  };
}

export default function SpeciesAdminEdit() {
  const nav = useNavigate();
  const { type, id } = useParams();

  const items = useSpeciesStore((s) => s.items);
  const upsert = useSpeciesStore((s) => s.upsert);

  const mode: "new" | "edit" = id ? "edit" : "new";
  const initialType: SpeciesType =
    (type as SpeciesType) || "animal";

  const existing = useMemo(() => {
    if (!type || !id) return null;
    return items.find((x) => x.type === type && x.id === id) || null;
  }, [items, type, id]);

  const [form, setForm] = useState<Species>(() => existing ?? emptySpecies(initialType));

  const [refs, setRefs] = useState<RefRow[]>(
    (existing?.references?.length ? existing.references : form.references).map((r) => ({
      title: r.title,
      url: r.url,
    }))
  );

  function set<K extends keyof Species>(key: K, value: Species[K]) {
    setForm((prev) => ({ ...prev, [key]: value }));
  }

  function onSave() {
    const cleanId = form.id.trim();
    if (!cleanId) return alert("id ห้ามว่าง");
    if (!/^[a-z0-9-]+$/.test(cleanId)) {
      return alert('id ต้องเป็น lowercase + ตัวเลข + "-" เท่านั้น เช่น call-duck');
    }
    if (!form.name_th.trim()) return alert("name_th ห้ามว่าง");
    if (!form.name_en.trim()) return alert("name_en ห้ามว่าง");
    if (!form.image.trim()) return alert("image ห้ามว่าง");

    const cleanRefs = refs
      .map((r) => ({ title: r.title.trim(), url: r.url.trim() }))
      .filter((r) => r.title && r.url);

    if (cleanRefs.length === 0) {
      return alert("ต้องมี reference อย่างน้อย 1 อัน");
    }

    const payload: Species = {
      ...form,
      id: cleanId,
      scientific_name: form.scientific_name?.trim() || undefined,
      references: cleanRefs,
    };

    upsert(payload);
    nav("/admin");
  }

  return (
    <div style={{ padding: 24, maxWidth: 980, margin: "0 auto" }}>
      <div style={{ display: "flex", alignItems: "center", gap: 12, flexWrap: "wrap" }}>
        <h1 style={{ margin: 0 }}>
          {mode === "new" ? "Add new species" : `Edit: ${form.type}/${form.id}`}
        </h1>

        <div style={{ marginLeft: "auto", display: "flex", gap: 10, flexWrap: "wrap" }}>
          <Link to="/admin" style={btnGhost}>← Back</Link>
          <button onClick={onSave} style={btnPrimary}>Save</button>
        </div>
      </div>

      <div style={{ marginTop: 16, display: "grid", gridTemplateColumns: "1fr 1fr", gap: 14 }}>
        {/* Type */}
        <div>
          <label style={label}>Type</label>
          <select
            value={form.type}
            onChange={(e) => {
              const t = e.target.value as SpeciesType;
              set("type", t);
              // auto suggest image folder
              set("image", t === "animal" ? "/images/animals/" : "/images/plants/");
            }}
            style={input}
            disabled={mode === "edit"} // กัน id/type เปลี่ยนแล้วชนกับของเดิม
          >
            <option value="animal">animal</option>
            <option value="plant">plant</option>
          </select>
          {mode === "edit" && (
            <div style={{ fontSize: 12, opacity: 0.7, marginTop: 6 }}>

            </div>
          )}
        </div>

        {/* ID */}
        <div>

            value={form.id}
            onChange={(e) => set("id", e.target.value)}
            placeholder='เช่น "call-duck"'

          />
        </div>

        {/* TH */}

            onChange={(e) => set("name_th", e.target.value)}
            style={input}
          />
        </div>

        {/* EN */}
        <div>

        {/* Scientific */}

    if (query) {
      list = list.filter((x) => {
        const th = (x.name_th || "").toLowerCase();
        const en = (x.name_en || "").toLowerCase();
        const sci = (x.scientific_name || "").toLowerCase();
        const id = (x.id || "").toLowerCase();
        return (
          th.includes(query) ||
          en.includes(query) ||
          sci.includes(query) ||
          id.includes(query)
        );
      });
    }

    // sort by type then name_en
    return [...list].sort((a, b) => {
      if (a.type !== b.type) return a.type.localeCompare(b.type);
      return (a.name_en || "").localeCompare(b.name_en || "");
    });
  }, [items, tab, q]);

          All ({counts.all})
        </button>

          Animals ({counts.animal})

          </thead>
          <tbody>
            {filtered.map((x) => (

                <td style={td}>

                </td>
                <td style={td} title={x.id}>
                  {x.id}
                </td>
                <td style={td}>{x.name_th}</td>
                <td style={td}>{x.name_en}</td>
                <td style={td}>
                  <div style={{ display: "flex", gap: 10, flexWrap: "wrap" }}>
                    <Link to={`/species/${x.type}/${x.id}`} style={linkBtn}>
                      View
                    </Link>
                    <Link to={`/admin/edit/${x.type}/${x.id}`} style={linkBtn}>
                      Edit
                    </Link>
                    <button
                      style={dangerBtn}
                      onClick={() => {
                        const ok = confirm(`Delete: ${x.type}/${x.id} ?`);
                        if (ok) remove(x.type, x.id);
                      }}
                    >
                      Delete
                    </button>
                  </div>
                </td>
              </tr>
            ))}

            {filtered.length === 0 && (
              <tr>
                <td style={{ ...td, padding: 18 }} colSpan={5}>
                  No data.
                </td>
              </tr>
            )}
          </tbody>
        </table>
      </div>
    </div>
  );
}

import { useMemo, useState } from "react";
import { useSpeciesStore } from "../store/speciesStore";
import type { Species, SpeciesType, Reference } from "../types/species";

type Mode = "create" | "edit" | null;

const emptyDraft = (): Species => ({
  id: "",
  type: "animal",
  name_th: "",
  name_en: "",
  scientific_name: "",
  short_description: "",
  description: "",
  image: "",
  tags: [],
  references: [],
});

export default function SpeciesManager() {
  const items = useSpeciesStore((s) => s.items);
  const add = useSpeciesStore((s) => s.add);
  const update = useSpeciesStore((s) => s.update);
  const remove = useSpeciesStore((s) => s.remove);

  const [query, setQuery] = useState("");
  const [typeFilter, setTypeFilter] = useState<"all" | SpeciesType>("all");

  const [mode, setMode] = useState<Mode>(null);
  const [editingKey, setEditingKey] = useState<{ type: SpeciesType; id: string } | null>(null);
  const [draft, setDraft] = useState<Species>(emptyDraft());

  const filtered = useMemo(() => {
    const q = query.trim().toLowerCase();
    return items
      .filter((x) => (typeFilter === "all" ? true : x.type === typeFilter))
      .filter((x) => {
        if (!q) return true;

        const refsText = (x.references ?? [])
          .map((r) => `${r.title ?? ""} ${r.url ?? ""}`)
          .join(" ");

        const hay = [
          x.id,
          x.name_th,
          x.name_en,
          x.scientific_name ?? "",
          x.short_description ?? "",
          (x.tags ?? []).join(" "),
          refsText,
        ]
          .join(" ")
          .toLowerCase();

        return hay.includes(q);
      })
      .sort((a, b) => (a.name_en || "").localeCompare(b.name_en || ""));
  }, [items, query, typeFilter]);

  function startCreate() {
    setMode("create");
    setEditingKey(null);
    setDraft(emptyDraft());
  }

  function startEdit(item: Species) {
    setMode("edit");
    setEditingKey({ type: item.type, id: item.id });
    setDraft({
      ...item,
      tags: item.tags ?? [],
      references: item.references ?? [],
      scientific_name: item.scientific_name ?? "",
      short_description: item.short_description ?? "",
      description: item.description ?? "",
    });
  }

  function cancel() {
    setMode(null);
    setEditingKey(null);
    setDraft(emptyDraft());
  }

  function validate(d: Species) {
    if (!d.id.trim()) return "กรุณาใส่ id (เช่น black-swan)";
    if (!d.name_th.trim()) return "กรุณาใส่ชื่อไทย";
    if (!d.name_en.trim()) return "กรุณาใส่ชื่ออังกฤษ";
    if (!d.image.trim()) return "กรุณาใส่ path รูป (เช่น /images/animals/xxx.jpg)";
    return "";
  }

  function onSave() {
    const err = validate(draft);
    if (err) {
      alert(err);
      return;
    }

    const normalized: Species = {
      ...draft,
      id: draft.id.trim(),
      name_th: draft.name_th.trim(),
      name_en: draft.name_en.trim(),
      image: draft.image.trim(),
      scientific_name: (draft.scientific_name ?? "").trim(),
      short_description: (draft.short_description ?? "").trim(),
      description: (draft.description ?? "").trim(),
      tags: (draft.tags ?? []).map((t) => t.trim()).filter(Boolean),
      references: (draft.references ?? [])
        .map((r) => ({
          title: (r.title ?? "").trim(),
          url: (r.url ?? "").trim(),
        }))
        .filter((r) => r.title || r.url),
    };

    if (mode === "create") {
      const dup = items.some((x) => x.type === normalized.type && x.id === normalized.id);
      if (dup) {
        alert("id นี้ซ้ำใน type เดียวกันแล้ว (เปลี่ยน id ก่อน)");
        return;
      }
      add(normalized);
      cancel();
      return;
    }

   if (mode === "edit" && editingKey) {
  update(editingKey.type, editingKey.id, normalized);
  cancel();
}
  }

  function onDelete(item: Species) {
    const ok = confirm(`ลบ: ${item.name_en} (${item.type}/${item.id}) ?`);
    if (!ok) return;
    remove(item.type, item.id);
    if (editingKey?.type === item.type && editingKey?.id === item.id) cancel();
  }

  function setReferenceLines(text: string) {
    // รูปแบบ: Title | URL ต่อ 1 บรรทัด
    const lines = text.split("\n").map((x) => x.trim()).filter(Boolean);
    const refs: Reference[] = lines.map((line) => {
      const [titleRaw, urlRaw] = line.split("|");
      return {
        title: (titleRaw ?? "").trim(),
        url: (urlRaw ?? "").trim(),
      };
    });
    setDraft((prev) => ({ ...prev, references: refs }));
  }

  const referencesText = (draft.references ?? [])
    .map((r) => `${r.title ?? ""} | ${r.url ?? ""}`.trim())
    .join("\n");

  return (
    <div style={{ maxWidth: 1100 }}>
      <div style={{ display: "flex", justifyContent: "space-between", gap: 12, flexWrap: "wrap" }}>
        <div>
          <h1 style={{ fontSize: 34, margin: 0, fontWeight: 900 }}>Species Manager</h1>
          <p style={{ marginTop: 8, color: "#475569" }}>
            CRUD จริง (แก้/เพิ่ม/ลบ) โดยเก็บใน store — หน้า Encyclopedia จะเปลี่ยนตามทันที
          </p>
        </div>

        <button
          onClick={startCreate}
          style={{
            padding: "10px 14px",
            borderRadius: 12,
            border: "1px solid #e2e8f0",
            background: "#0f172a",
            color: "#fff",
            fontWeight: 900,
            cursor: "pointer",
            height: 44,
          }}
        >
          + New species
        </button>
      </div>

      {/* Filters */}
      <div style={{ display: "flex", gap: 10, flexWrap: "wrap", marginTop: 12 }}>
        <select
          value={typeFilter}
          onChange={(e) => setTypeFilter(e.target.value as "all" | SpeciesType)}
          style={{ padding: 10, borderRadius: 12, border: "1px solid #e2e8f0" }}
        >
          <option value="all">All</option>
          <option value="animal">Animals</option>
          <option value="plant">Plants</option>
        </select>

        <input
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          placeholder="Search..."
          style={{
            flex: 1,
            minWidth: 240,
            padding: 10,
            borderRadius: 12,
            border: "1px solid #e2e8f0",
          }}
        />

        <div style={{ alignSelf: "center", color: "#64748b", fontWeight: 800 }}>
          {filtered.length} items
        </div>
      </div>

      {/* Editor */}
      {mode && (
        <div
          style={{
            marginTop: 14,
            background: "#fff",
            border: "1px solid #e2e8f0",
            borderRadius: 16,
            padding: 14,
          }}
        >
          <div style={{ display: "flex", justifyContent: "space-between", gap: 10, flexWrap: "wrap" }}>
            <h2 style={{ margin: 0 }}>{mode === "create" ? "Create" : "Edit"}</h2>
            <div style={{ display: "flex", gap: 10 }}>
              <button onClick={onSave} style={btnPrimary}>
                Save
              </button>
              <button onClick={cancel} style={btnGhost}>
                Cancel
              </button>
            </div>
          </div>

          <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: 10, marginTop: 12 }}>
            <label style={labelStyle}>
              Type
              <select
                value={draft.type}
                onChange={(e) => setDraft({ ...draft, type: e.target.value as SpeciesType })}
                disabled={mode === "edit"}
                style={inputStyle}
              >
                <option value="animal">animal</option>
                <option value="plant">plant</option>
              </select>
            </label>

            <label style={labelStyle}>
              ID (url key)

                value={draft.id}

                disabled={mode === "edit"}

  return <Outlet />;
}

import React, { createContext, useContext, useEffect, useMemo, useState } from "react";
import type { User } from "firebase/auth";
import { onAuthStateChanged, signOut } from "firebase/auth";
import { auth } from "../services/firebase";
import { getRoleByEmail, type Role } from "../utils/roles";

type AuthState = {
  user: User | null;
  role: Role;
  loading: boolean;
  logout: () => Promise<void>;
};

const AuthContext = createContext<AuthState | null>(null);

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const unsub = onAuthStateChanged(auth, (u) => {
      setUser(u);
      setLoading(false);
    });
    return () => unsub();
  }, []);

  const role = useMemo(() => getRoleByEmail(user?.email), [user?.email]);

  async function logout() {
    await signOut(auth);
  }

  const value: AuthState = { user, role, loading, logout };
  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export function useAuth() {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error("useAuth must be used within AuthProvider");
  return ctx;
}

import { create } from "zustand";
import type { Species, SpeciesType } from "../types/species";
import { animals } from "../data/animals";
import { plants } from "../data/plants";

type MergeResult = {
  ok: boolean;
  error?: string;
  totalInFile?: number;
  added?: number;
  skipped?: number;
  invalid?: number;
  countAfter?: number;
};

type SpeciesState = {
  items: Species[];
  updatedAt: number;

  setItems: (items: Species[]) => void;
  add: (item: Species) => void;
  update: (type: SpeciesType, id: string, next: Species) => void;
  remove: (type: SpeciesType, id: string) => void;

  hydrate: () => void;
  resetToSeed: () => void;

  // ✅ Backup/Restore
  exportJSON: () => string;
  importJSON: (jsonText: string) => { ok: boolean; error?: string; count?: number };

  // ✅ Merge Import (รวม ไม่ทับ)
  mergeImportJSON: (jsonText: string) => MergeResult;
};

const STORAGE_KEY = "udomtong.speciesStore.v1";

// 👇 รวม animals + plants เป็น seed เริ่มต้น
const SEED_ITEMS: Species[] = [...animals, ...plants];

function safeParse<T>(raw: string | null): T | null {
  if (!raw) return null;
  try {
    return JSON.parse(raw) as T;
  } catch {
    return null;
  }
}

function now() {
  return Date.now();
}

function normalize(items: Species[]): Species[] {
  return (items ?? []).map((x) => ({
    ...x,
    scientific_name: x.scientific_name ?? "",
    short_description: x.short_description ?? "",
    description: x.description ?? "",
    tags: x.tags ?? [],
    references: x.references ?? [],
  }));
}

function save(items: Species[], updatedAt: number) {
  localStorage.setItem(
    STORAGE_KEY,
    JSON.stringify({
      items,
      updatedAt,
    })
  );
}

function load(): { items: Species[]; updatedAt: number } | null {
  const raw = localStorage.getItem(STORAGE_KEY);
  const parsed = safeParse<{ items: Species[]; updatedAt: number }>(raw);
  if (!parsed?.items) return null;

  return {
    items: normalize(parsed.items),
    updatedAt: parsed.updatedAt ?? now(),
  };
}

function isValidSpecies(it: any): it is Species {
  return !!(
    it &&
    typeof it.id === "string" &&
    typeof it.type === "string" &&
    typeof it.name_th === "string" &&
    typeof it.name_en === "string" &&
    typeof it.image === "string" &&
    it.id.trim() &&
    it.type.trim() &&
    it.name_th.trim() &&
    it.name_en.trim() &&
    it.image.trim()
  );
}

function extractItemsFromJSON(parsed: any): any[] | null {
  // รองรับ 2 แบบ:
  // 1) เป็น array ตรงๆ
  // 2) เป็น object { items: [...] }
  if (Array.isArray(parsed)) return parsed;
  if (Array.isArray(parsed?.items)) return parsed.items;
  return null;
}

function makeKey(type: string, id: string) {
  return `${String(type).toLowerCase()}:${String(id).toLowerCase()}`;
}

export const useSpeciesStore = create<SpeciesState>((set, get) => {
  const stored = typeof window !== "undefined" ? load() : null;

  const initialItems =
    stored?.items?.length ? stored.items : normalize(SEED_ITEMS);

  const initialUpdatedAt = stored?.updatedAt ?? now();

  // ถ้ายังไม่มี storage ให้เซฟ seed ลงไปเลย
  if (typeof window !== "undefined" && !stored) {
    save(initialItems, initialUpdatedAt);
  }

  return {
    items: initialItems,
    updatedAt: initialUpdatedAt,

    setItems: (items) => {
      const next = normalize(items);
      const ts = now();
      set({ items: next, updatedAt: ts });
      save(next, ts);
    },

    add: (item) => {
      const state = get();
      const next = normalize([...state.items, item]);
      const ts = now();
      set({ items: next, updatedAt: ts });
      save(next, ts);
    },

    update: (type, id, nextItem) => {
      const state = get();
      const next = normalize(
        state.items.map((x) =>
          x.type === type && x.id === id ? nextItem : x
        )
      );
      const ts = now();
      set({ items: next, updatedAt: ts });
      save(next, ts);
    },

    remove: (type, id) => {
      const state = get();
      const next = normalize(
        state.items.filter(
          (x) => !(x.type === type && x.id === id)
        )
      );
      const ts = now();
      set({ items: next, updatedAt: ts });
      save(next, ts);
    },

    hydrate: () => {
      const storedNow = load();
      if (!storedNow) return;
      set({
        items: storedNow.items,
        updatedAt: storedNow.updatedAt,
      });
    },

    resetToSeed: () => {
      const next = normalize(SEED_ITEMS);
      const ts = now();
      set({ items: next, updatedAt: ts });
      save(next, ts);
    },

    // ✅ Export/Import JSON
    exportJSON: () => {
      const state = get();
      return JSON.stringify(
        {
          version: 1,
          exportedAt: new Date().toISOString(),
          items: state.items,
        },
        null,
        2
      );
    },

    importJSON: (jsonText: string) => {
      try {
        const parsed = JSON.parse(jsonText);
        const items = extractItemsFromJSON(parsed);

        if (!items) {
          return { ok: false, error: "ไฟล์ JSON ไม่ถูกต้อง (ต้องเป็น array หรือมี field items)" };
        }

        const normalized = normalize(items as Species[]);

        for (const it of normalized) {
          if (!isValidSpecies(it)) {
            return {
              ok: false,
              error: "ข้อมูลบางรายการขาด field จำเป็น (id/type/name_th/name_en/image)",
            };
          }
        }

        const ts = now();
        set({ items: normalized, updatedAt: ts });
        save(normalized, ts);

        return { ok: true, count: normalized.length };
      } catch (e: any) {
        return { ok: false, error: e?.message || "JSON parse error" };
      }
    },

    // ✅ Merge Import JSON (รวม ไม่ทับของเดิม)
    mergeImportJSON: (jsonText: string) => {
      try {
        const parsed = JSON.parse(jsonText);
        const incoming = extractItemsFromJSON(parsed);

        if (!incoming) {
          return {
            ok: false,
            error: "ไฟล์ JSON ไม่ถูกต้อง (ต้องเป็น array หรือมี field items)",
          };
        }

        const state = get();
        const existing = state.items ?? [];
        const existingMap = new Set(existing.map((x) => makeKey(x.type, x.id)));

        let added = 0;
        let skipped = 0;
        let invalid = 0;

        const toAdd: Species[] = [];

        for (const raw of incoming) {
          const candidate = normalize([raw as Species])[0];

          if (!isValidSpecies(candidate)) {
            invalid++;
            continue;
          }

          const key = makeKey(candidate.type, candidate.id);
          if (existingMap.has(key)) {
            skipped++;
            continue;
          }

          existingMap.add(key);
          toAdd.push(candidate);
          added++;
        }

        // ไม่มีอะไรเพิ่มก็ไม่ต้องเซฟ
        if (added === 0) {
          return {
            ok: true,
            totalInFile: incoming.length,
            added,
            skipped,
            invalid,
            countAfter: existing.length,
          };
        }

        const next = normalize([...existing, ...toAdd]);
        const ts = now();
        set({ items: next, updatedAt: ts });
        save(next, ts);

        return {
          ok: true,
          totalInFile: incoming.length,
          added,
          skipped,
          invalid,
          countAfter: next.length,
        };
      } catch (e: any) {
        return { ok: false, error: e?.message || "JSON parse error" };
      }
    },
  };
});
